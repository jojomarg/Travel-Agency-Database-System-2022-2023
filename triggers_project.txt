USE travel_agency;


DROP TABLE log;
CREATE TABLE log(
Operation VARCHAR(20),
UpdatedOn DATETIME,
UpdatedBy VARCHAR(20));


DROP TRIGGER IF EXISTS loginsert;
DELIMITER $
CREATE TRIGGER loginsert
AFTER INSERT ON trip
FOR EACH ROW
BEGIN
INSERT INTO log(Operation,UpdatedOn,UpdatedBy)
VALUES ('INSERT',CURDATE(),SUSER_NAME());
END$
DELIMITER ;
