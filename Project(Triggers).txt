ημερομηνίας επιστροφής και του κόστους του ταξιδιού, αν έχουν ήδη γίνει κρατήσεις
γι’ αυτό. 

DROP TRIGGER IF EXISTS check_trip;
DELIMITER $
CREATE TRIGGER check_trip BEFORE UPDATE ON trip
FOR EACH ROW
BEGIN
DECLARE reservation_id INT;
SELECT reservation.res_tr_id INTO reservation_id FROM reservation INNER JOIN trip ON reservation.res_tr_id=trip.tr_id LIMIT 0,1;
IF(tr_id=reservation_id) THEN
SET NEW.tr_departure=OLD.tr_departure;
SET NEW.tr_return=OLD.tr_return;
SET NEW.tr_cost=OLD.tr_cost;
END IF;
END$

CREATE TABLE trip(
tr_id INT(11) NOT NULL AUTO_INCREMENT,						/* είναι κωδικός οπότε θέλουμε πάντα να έχει τιμή και να δίνεται αυτόματα με αυξηση της τιμής καθε φορά */
tr_departure DATETIME,
tr_return DATETIME,
tr_maxseats TINYINT(4) NOT NULL,
tr_cost FLOAT(7,2) NOT NULL,						/* ΔΕΝ ΕΙΜΑΙ ΣΙΓΟΥΡΗ ΑΝ ΓΡΑΦΕΤΑΙ ΕΤΣΙ */
tr_br_code INT(11) NOT NULL,								/* κλειδί της branch οντότητας */
tr_gui_AT CHAR(10) NOT NULL,								/* κλειδί guide οντότητας */
tr_drv_AT CHAR(10) NOT NULL,								/* κλειδί driver οντότητας */
PRIMARY KEY(tr_id),
FOREIGN KEY(tr_br_code) REFERENCES branch(br_code)		/* νομιζω πωσ γράφεται ετσι, αφού το βελάκι φεύγει από την οντότητα trip, συγκεκριμένα το γνώρισμα tr_br_code και πηγαίνει στην οντότητα branch συγκεκριένα στο γνώρισμα και κλειδί br_code */
ON DELETE CASCADE ON UPDATE CASCADE,
FOREIGN KEY(tr_gui_AT) REFERENCES guide(gui_AT)		/* νομιζω πωσ γράφεται ετσι, αφού το βελάκι φεύγει από την οντότητα trip, συγκεκριμένα το γνώρισμα tr_gui_AT και πηγαίνει στην οντότητα guide συγκεκριένα στο γνώρισμα και κλειδί gui_AT */
ON DELETE CASCADE ON UPDATE CASCADE,
FOREIGN KEY(tr_drv_AT) REFERENCES driver(drv_AT)		/* νομιζω πωσ γράφεται ετσι, αφού το βελάκι φεύγει από την οντότητα trip, συγκεκριμένα το γνώρισμα tr_drv_AT και πηγαίνει στην οντότητα driver συγκεκριένα στο γνώρισμα και κλειδί tr_drv_AT */
ON DELETE CASCADE ON UPDATE CASCADE
);

CREATE TABLE reservation(
res_tr_id INT(11) NOT NULL,									/* κλειδί trip οντότητας */
res_seatnum TINYINT(4) NOT NULL,				/* είναι κωδικός οπότε θέλουμε πάντα να έχει τιμή */
res_name VARCHAR(20) DEFAULT 'unknown' NOT NULL, 			/* αν δεν έχει τιμή by default θα πάρει την τιμή unknown */
res_lname VARCHAR(20) DEFAULT 'unknown' NOT NULL, 			/* αν δεν έχει τιμή by default θα πάρει την τιμή unknown */
res_isadult ENUM('ADULT', 'MINOR') NOT NULL,							/* μπορεί να πάρει τιμές μεταξύ ADULT και MINOR */
PRIMARY KEY(res_tr_id, res_seatnum),
FOREIGN KEY(res_tr_id) REFERENCES trip(tr_id)				/* συνδέονται με το κλειδί tr_id */
ON DELETE CASCADE ON UPDATE CASCADE
);
